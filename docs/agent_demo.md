# Agent demo: governed metrics over MCP

Trace of the four MCP tools answering *What happened to churn after the July 2025 price increase?*
Generated by `scripts/agent_demo.py` from the default dataset; every output below is real.

## 1. `list_metrics`

30 metrics: 7 ratio, 18 flow, 1 derived, 4 snapshot.

## 2. `describe_metric`

- **logo_churn_rate** (ratio): Customers lost in the period over paying customers at the start of the period. Definition: `churned_customers / opening_customers_metric`.
- **expansion_mrr** (flow): MRR increase from existing customers (price increases, more subscriptions). Flow. Definition: `{{ Dimension('customer_month__movement_type') }} = 'expansion'`.

  SELECT
    DATE_TRUNC('month', month_start) AS mrr_month__month
    , DATE_TRUNC('month', month_start) AS metric_time__month
    , movement_type AS customer_month__movement_type
    , case when previous_mrr_usd > 0 then 1 else 0 end AS __opening_customers_metric
    , mrr_change_usd AS __churned_mrr
    , mrr_change_usd AS __expansion_mrr
    , case when movement_type = 'churn' then 1 else 0 end AS __churned_customers
  FROM "warehouse"."finance"."fct_mrr_monthly" customer_mrr_src_10000
)

, cm_5_cte AS (
  SELECT
    subq_3.metric_time__month AS metric_time__month
    , SUM(subq_3.__opening_customers_metric) AS opening_customers_metric
  FROM (
    SELECT
      opening_customers_metric AS __opening_customers_metric
      , mrr_month__month
      , metric_time__month
    FROM (
      SELECT
        mrr_month__month
        , metric_time__month
        , __opening_customers_metric AS opening_customers_metric
      FROM sma_10002_cte

    WHERE metric_time__month between '2025-04-01' and '2025-10-01'
  ) subq_3
  INNER JOIN (
    SELECT
      metric_time__month
      , MIN(mrr_month__month) AS mrr_month__month__complete
    FROM (
      SELECT
        mrr_month__month
        , metric_time__month
      FROM sma_10002_cte
    ) subq_2
    WHERE metric_time__month between '2025-04-01' and '2025-10-01'
    GROUP BY
      metric_time__month
  ) subq_5
  ON
    subq_3.mrr_month__month = subq_5.mrr_month__month__complete
  GROUP BY
    subq_3.metric_time__month
)

, cm_6_cte AS (

    metric_time__month
    , COALESCE(__churned_customers, 0) AS churned_customers
  FROM (

      metric_time__month
      , SUM(churned_customers) AS __churned_customers
    FROM (
      SELECT
        metric_time__month

      FROM sma_10002_cte

    GROUP BY

SELECT

  , MAX(subq_9.opening_customers_metric) AS opening_customers_metric
  , COALESCE(MAX(subq_15.churned_customers), 0) AS churned_customers
  , MAX(subq_19.logo_churn_rate) AS logo_churn_rate
  , COALESCE(MAX(subq_25.expansion_mrr), 0) AS expansion_mrr
  , COALESCE(MAX(subq_31.churned_mrr), 0) AS churned_mrr
FROM (
  SELECT
    metric_time__month
    , opening_customers_metric
  FROM cm_5_cte
) subq_9
FULL OUTER JOIN (
  SELECT
    metric_time__month
    , churned_customers
  FROM cm_6_cte
) subq_15
ON
  subq_9.metric_time__month = subq_15.metric_time__month
FULL OUTER JOIN (
  SELECT
    metric_time__month
    , CAST(churned_customers AS DOUBLE) / CAST(NULLIF(opening_customers_metric, 0) AS DOUBLE) AS logo_churn_rate
  FROM (
    SELECT
      COALESCE(subq_16.metric_time__month, subq_17.metric_time__month) AS metric_time__month
      , COALESCE(MAX(subq_16.churned_customers), 0) AS churned_customers

    FROM (

        metric_time__month

        metric_time__month

      FROM cm_5_cte
    ) subq_17
    ON
      subq_16.metric_time__month = subq_17.metric_time__month
    GROUP BY
      COALESCE(subq_16.metric_time__month, subq_17.metric_time__month)
  ) subq_18

ON
  COALESCE(subq_9.metric_time__month, subq_15.metric_time__month) = subq_19.metric_time__month

    metric_time__month

      metric_time__month

      metric_time__month
